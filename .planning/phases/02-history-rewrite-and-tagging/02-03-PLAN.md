1. Run the exhaustive blob scan -- enumerate ALL objects in the rewritten history and check for surviving sensitive content:

```bash
#!/bin/bash
echo "=== BLOB SCAN: Checking for surviving sensitive content ==="
FAIL_COUNT=0

git rev-list --all --objects | while read hash path; do
  if [ -n "$path" ]; then
    # Check targeted extensions
    case "$path" in
      *.pdf|*.sqlite|*.key|*.jsonl|*.tar.gz)
        echo "FAIL: $path ($hash)"
        FAIL_COUNT=$((FAIL_COUNT + 1))
        ;;
    esac
    # Check .docx -- only two templates should survive
    case "$path" in
      *.docx)
        case "$path" in
          */cv_template_wcm.docx|*/wcm_cv_template_faculty_october_2022_final.docx)
            echo "OK (template): $path"
            ;;
          *)
            echo "FAIL: unexpected .docx: $path ($hash)"
            FAIL_COUNT=$((FAIL_COUNT + 1))
            ;;
        esac
        ;;
    esac
    # Check .db files
    case "$path" in
      *.db)
        echo "FAIL: $path ($hash)"
        FAIL_COUNT=$((FAIL_COUNT + 1))
        ;;
    esac
    # Check sensitive directory paths
    case "$path" in
      data/*|archive/*|*/node_modules/*|*/uploads/*|*/prompt_logs/*|*/prompt_logs_debug/*|*/dist/*|*/outputs/*|src/logs/*)
        echo "FAIL: $path ($hash)"
        FAIL_COUNT=$((FAIL_COUNT + 1))
        ;;
    esac
    # Check specific sensitive files
    case "$path" in
      */auth_config.yaml|*/cviche.db|*/cviche_dev.db|*/pipeline.db)
        echo "FAIL: $path ($hash)"
        FAIL_COUNT=$((FAIL_COUNT + 1))
        ;;
    esac
  fi
done

echo ""
echo "=== RESULTS ==="
echo "Failures: $FAIL_COUNT"
```
