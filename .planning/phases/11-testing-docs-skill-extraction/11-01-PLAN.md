```python
"""Integration tests for SAML flow against mock IdP container.

Requires: docker compose -f docker-compose.mock-idp.yml up -d
Tests skip gracefully when mock IdP is not running.
"""
import pytest
import httpx
from unittest.mock import patch, MagicMock
import os

MOCK_IDP_URL = "http://localhost:8443"
MOCK_IDP_METADATA = f"{MOCK_IDP_URL}/simplesaml/saml2/idp/metadata.php"

def _mock_idp_available() -> bool:
    try:
        r = httpx.get(f"{MOCK_IDP_URL}/simplesaml/", timeout=2)
        return r.status_code == 200
    except (httpx.ConnectError, httpx.ReadTimeout):
        return False

requires_mock_idp = pytest.mark.skipif(
    not _mock_idp_available(),
    reason="Mock IdP not running (docker compose -f docker-compose.mock-idp.yml up -d)"
)
```

   These are the key end-to-end tests. The pattern is:
   1. Use httpx to GET /api/saml/login (follow redirect to mock IdP)
   2. Programmatically POST credentials to the mock IdP's login form
   3. Capture the SAMLResponse from the IdP's auto-submit form
   4. POST the SAMLResponse to /api/saml/acs
   5. Verify session cookie is set

   **IMPORTANT:** This is the hardest test to get right. The flow involves multiple redirects and form submissions. If programmatic IdP login proves too complex with httpx (IdP login forms vary), fall back to testing the ACS endpoint directly by:

   - `test_acs_expired_cache_ed_down`: User with expired cache and ED unavailable -> 401 on /api/auth/me

   These tests use `unittest.mock.patch` on `get_saml_client` and `check_ed_membership` (same pattern as existing tests in test_ed_group.py and test_saml_sp.py), but exercise the full request->response cycle including config seeding and cookie handling.
